📋 Lists and Tuples

Lists (Mutable)

⚡ Functions

# Basic function
def greet(name):
    return f"Hello, {name}!"

# Function with default parameters

# *args and **kwargs
def sum_all(*args):
    return sum(args)

print(sum_all(1, 2, 3, 4))  # 10

def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

print_info(name="Alice", age=30)

# Lambda functions (anonymous functions)
square = lambda x: x**2
print(square(5))  # 25

# Map, Filter, Reduce
numbers = [1, 2, 3, 4, 5]
squared = list(map(lambda x: x**2, numbers))
evens = list(filter(lambda x: x % 2 == 0, numbers))

# Decorators (used heavily in Flask!)
def uppercase_decorator(func):
    def wrapper(*args, **kwargs):
        result = func(*args, **kwargs)
        return result.upper()
    return wrapper

@uppercase_decorator
def greet():
    return "hello world"

print(greet())  # HELLO WORLD

🏗️ Object-Oriented Programming (OOP)

OOP is a programming paradigm based on the concept of "objects", which can contain data (attributes) and code (methods). Python is a powerful object-oriented language.

The Four Pillars of OOP:

  • Encapsulation: Bundling data and methods that work on that data within one unit (a class).
  • Inheritance: A way to form new classes using classes that have already been defined.
  • Polymorphism: The ability of an object to take on many forms. The most common use is method overriding.
  • Abstraction: Hiding complex implementation details and showing only the necessary features of an object.
# Define a class
class Person:
    # Class variable (shared by all instances)
    species = "Homo sapiens"
    
    # Constructor
    def __init__(self, name, age):
        # Instance variables
        self.name = name
        self.age = age
    
    # Instance method
    def greet(self):
        return f"Hello, I'm {self.name}"
    
    # Method with parameters
    def have_birthday(self):
        self.age += 1
        return f"Happy birthday! Now {self.age} years old"
    
    # String representation
    def __str__(self):
        return f"Person(name={self.name}, age={self.age})"

# Create objects
person1 = Person("Alice", 30)
person2 = Person("Bob", 25)

print(person1.greet())           # Hello, I'm Alice
print(person1.have_birthday())   # Happy birthday! Now 31 years old
print(person1)                   # Person(name=Alice, age=31)

# Inheritance
class Student(Person):
    def __init__(self, name, age, student_id):
        super().__init__(name, age)  # Call parent constructor
        self.student_id = student_id
    
    def study(self):
        return f"{self.name} is studying"
    
    # Override parent method
    def greet(self):
        return f"Hi, I'm {self.name}, student ID: {self.student_id}"

student = Student("Charlie", 20, "S123")
print(student.greet())   # Hi, I'm Charlie, student ID: S123
print(student.study())   # Charlie is studying

# Properties (getters and setters)
class User:
    def __init__(self, name):
        self._name = name  # Protected attribute
    
    @property
    def name(self):
        return self._name
    
    @name.setter
    def name(self, value):
        if len(value) < 2:
            raise ValueError("Name too short")
        self._name = value

user = User("Alice")
print(user.name)      # Alice
user.name = "Bob"     # Uses setter

🛡️ Encapsulation

Encapsulation restricts direct access to an object's components. In Python, this is done by convention using name prefixes.

class Car:
    def __init__(self, make, model):
        self.make = make              # Public attribute
        self._speed = 0               # Protected attribute (convention)
        self.__engine_status = "Off"  # Private attribute (name mangling)

    def start_engine(self):
        self.__engine_status = "On"
        return "Engine started."

    def stop_engine(self):
        self.__engine_status = "Off"
        self._speed = 0
        return "Engine stopped."

    def accelerate(self):
        if self.__engine_status == "On":
            self._speed += 10
            return f"Accelerating to {self._speed} km/h"
        return "Engine is off!"

my_car = Car("Toyota", "Camry")
print(my_car.make)  # Accessible

# Accessing protected attribute (by convention, you shouldn't do this directly)
# print(my_car._speed)

# Cannot access private attribute directly
# print(my_car.__engine_status)  # This will raise an AttributeError

print(my_car.start_engine())
print(my_car.accelerate())

# Accessing "private" attribute via name mangling (for demonstration only)
print(my_car._Car__engine_status) # On

🎭 Polymorphism

Polymorphism allows objects of different classes to be treated as objects of a common superclass. It's about using a single interface to represent different underlying forms (data types).

class Dog:
    def speak(self):
        return "Woof!"

class Cat:
    def speak(self):
        return "Meow!"

class Duck:
    def speak(self):
        return "Quack!"

# A function that can work with any object that has a 'speak' method
def make_animal_speak(animal):
    print(animal.speak())

dog = Dog()
cat = Cat()
duck = Duck()

make_animal_speak(dog)   # Woof!
make_animal_speak(cat)   # Meow!
make_animal_speak(duck)  # Quack!

# This is known as "Duck Typing": if it walks like a duck and quacks like a duck, it's a duck.

👻 Abstraction

Abstraction hides the complex reality while exposing only the essential parts. In Python, we can use Abstract Base Classes (ABCs).

from abc import ABC, abstractmethod

# Abstract Base Class
class Vehicle(ABC):
    @abstractmethod
    def start(self):
        pass

    @abstractmethod
    def stop(self):
        pass

# Concrete class implementing the abstract class
class Car(Vehicle):
    def start(self):
        print("Car engine started.")

    def stop(self):
        print("Car engine stopped.")

class Motorcycle(Vehicle):
    def start(self):
        print("Motorcycle engine started.")

    def stop(self):
        print("Motorcycle engine stopped.")

# You cannot create an instance of an abstract class
# vehicle = Vehicle() # This would raise a TypeError

car = Car()
motorcycle = Motorcycle()

car.start()
motorcycle.start()

✨ Magic (Dunder) Methods

Dunder (double underscore) methods allow you to emulate the behavior of built-in types. They are hooks that let you customize Python's behavior.

class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    # String representation for developers
    def __repr__(self):
        return f"Vector({self.x}, {self.y})"

    # String representation for users
    def __str__(self):
        return f"({self.x}, {self.y})"

    # Overload the + operator
    def __add__(self, other):
        return Vector(self.x + other.x, self.y + other.y)

    # Overload the len() function
    def __len__(self):
        return 2

v1 = Vector(2, 3)
v2 = Vector(3, 4)

print(v1)          # (2, 3) - uses __str__
print(repr(v1))    # Vector(2, 3) - uses __repr__
print(v1 + v2)     # (5, 7) - uses __add__
print(len(v1))     # 2 - uses __len__

@ Static and Class Methods

Methods can belong to an instance, the class, or be independent.

class MyClass:
    class_var = "I am a class variable"

    def __init__(self, instance_var):
        self.instance_var = instance_var

    # Regular instance method (needs 'self')
    def instance_method(self):
        return f"Instance method called for {self.instance_var}"

    # Class method (needs 'cls')
    @classmethod
    def class_method(cls):
        return f"Class method called. Class var: {cls.class_var}"

    # Static method (no 'self' or 'cls')
    @staticmethod
    def static_method(x, y):
        return f"Static method called with {x} and {y}. It's just a function."

# Instance method needs an object
obj = MyClass("my_instance")
print(obj.instance_method())

# Class method can be called on the class or an instance
print(MyClass.class_method())
print(obj.class_method())

# Static method can be called on the class or an instance
print(MyClass.static_method(1, 2))
print(obj.static_method(3, 4))

⚠️ Exception Handling

# Try-except
try:
    result = 10 / 0
except ZeroDivisionError:
    print("Cannot divide by zero!")

# Multiple exceptions
try:
    number = int(input("Enter a number: "))
    result = 10 / number
except ValueError:
    print("Invalid input!")
except ZeroDivisionError:
    print("Cannot divide by zero!")
except Exception as e:
    print(f"An error occurred: {e}")

# Finally block (always executes)
try:
    file = open('file.txt', 'r')
    content = file.read()
except FileNotFoundError:
    print("File not found!")
finally:
    file.close()  # Always close the file

# Raise exceptions
def divide(a, b):
    if b == 0:
        raise ValueError("Denominator cannot be zero")
    return a / b

✨ Python Best Practices

  • PEP 8: Follow Python's style guide (use 4 spaces for indentation)
  • Naming: Use snake_case for variables and functions, PascalCase for classes
  • Comments: Use # for single-line, """...""" for docstrings
  • List Comprehensions: Use when it improves readability
  • Context Managers: Always use `with` for file operations
  • Exception Handling: Be specific about exceptions you catch
  • Type Hints: Use them for better code documentation (Python 3.5+)
  • Virtual Environments: Always use venv or conda for projects
